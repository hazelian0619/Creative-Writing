### 1.2 架构设计原则

1. **微服务架构**: 服务解耦，独立部署和扩展
2. **事件驱动**: 异步处理，提高系统响应性
3. **容器化部署**: Docker + Kubernetes，便于运维管理
4. **数据一致性**: 最终一致性，支持分布式事务
5. **高可用设计**: 故障自愈，无单点故障

```python
from fastapi import FastAPI, HTTPException, BackgroundTasks
from pydantic import BaseModel
import asyncio
import redis.asyncio as redis
from typing import Dict, List, Any

class AIInferenceService:
    def __init__(self):
        self.app = FastAPI(
            title="AI推理服务",
            description="处理DIFCM、模糊推理、GraphSAGE的核心AI算法",
            version="1.0.0"
        )
        
        # 算法引擎初始化
        self.difcm_engine = None
        self.fuzzy_engine = None
        self.graphsage_engine = None
        self.coordinator = None
        
        # 缓存和队列
        self.redis_client = None
        self.inference_queue = asyncio.Queue(maxsize=1000)
        
        # 注册路由
        self._register_routes()
        
    async def startup(self):
        """服务启动初始化"""
        # 初始化Redis连接
        self.redis_client = redis.Redis(
            host="redis-service", 
            port=6379, 
            decode_responses=True
        )
        
        # 初始化AI算法引擎
        await self._initialize_ai_engines()
        
        # 启动推理队列处理器
        asyncio.create_task(self._process_inference_queue())
        
    async def _initialize_ai_engines(self):
        """初始化AI算法引擎"""
        from ai_algorithms import DynamicIntervalFCM, TSFuzzyInferenceEngine, CreativeWritingGraphSAGE, AlgorithmCoordinator
        
        print("正在初始化AI算法引擎...")
        
        # 初始化DIFCM引擎
        self.difcm_engine = DynamicIntervalFCM()
        
        # 初始化模糊推理引擎
        self.fuzzy_engine = TSFuzzyInferenceEngine()
        
        # 初始化GraphSAGE引擎
        self.graphsage_engine = CreativeWritingGraphSAGE()
        await self.graphsage_engine.load_model("models/graphsage_creative_writing.pt")
        
        # 初始化算法协调器
        self.coordinator = AlgorithmCoordinator()
        
        print("AI算法引擎初始化完成")
    
    def _register_routes(self):
        """注册API路由"""
        
        @self.app.post("/inference/process")
        async def process_user_interaction(request: UserInteractionRequest):
            """处理用户交互推理请求"""
            try:
                # 添加到推理队列
                inference_task = InferenceTask(
                    request_id=request.request_id,
                    user_id=request.user_id,
                    input_data=request.input_data,
                    context_history=request.context_history
                )
                
                await self.inference_queue.put(inference_task)
                
                return {"request_id": request.request_id, "status": "queued"}
                
            except Exception as e:
                raise HTTPException(status_code=500, detail=str(e))
        
        @self.app.get("/inference/result/{request_id}")
        async def get_inference_result(request_id: str):
            """获取推理结果"""
            try:
                # 从Redis缓存获取结果
                result = await self.redis_client.get(f"inference_result:{request_id}")
                
                if result is None:
                    return {"status": "processing"}
                
                return {"status": "completed", "result": json.loads(result)}
                
            except Exception as e:
                raise HTTPException(status_code=500, detail=str(e))
        
        @self.app.get("/health")
        async def health_check():
            """健康检查"""
            return {
                "status": "healthy",
                "engines": {
                    "difcm": self.difcm_engine is not None,
                    "fuzzy": self.fuzzy_engine is not None,
                    "graphsage": self.graphsage_engine is not None
                },
                "queue_size": self.inference_queue.qsize()
            }
    
    async def _process_inference_queue(self):
        """处理推理队列"""
        while True:
            try:
                # 从队列获取推理任务
                task = await self.inference_queue.get()
                
                # 执行推理
                result = await self._execute_inference(task)
                
                # 缓存结果
                await self.redis_client.setex(
                    f"inference_result:{task.request_id}",
                    300,  # 5分钟过期
                    json.dumps(result)
                )
                
                # 标记任务完成
                self.inference_queue.task_done()
                
            except Exception as e:
                print(f"推理队列处理错误: {e}")
                await asyncio.sleep(1)
    
    async def _execute_inference(self, task: InferenceTask) -> Dict[str, Any]:
        """执行AI推理"""
        start_time = time.time()
        
        try:
            # 1. DIFCM认知状态更新
            cognitive_state = await self._run_difcm_inference(
                task.user_id, task.input_data, task.context_history
            )
            
            # 2. 模糊推理决策
            fuzzy_result = await self._run_fuzzy_inference(cognitive_state)
            
            # 3. GraphSAGE知识推理
            graph_result = await self._run_graphsage_inference(
                task.input_data, cognitive_state
            )
            
            # 4. 算法协调
            final_result = self.coordinator.coordinate_output(
                graph_result['layered_prompts'],
                fuzzy_result['prompt_intensity'],
                fuzzy_result['prompt_type'],
                fuzzy_result['calibration_level']
            )
            
            inference_time = time.time() - start_time
            
            return {
                "ai_response": final_result,
                "cognitive_state": cognitive_state,
                "inference_time": inference_time,
                "metadata": {
                    "fuzzy_result": fuzzy_result,
                    "graph_stats": graph_result['stats']
                }
            }
            
        except Exception as e:
            return {
                "error": str(e),
                "inference_time": time.time() - start_time
            }
```

```python
import httpx
import asyncio
from typing import Optional, Dict, Any

class ServiceClient:
    """服务间HTTP通信客户端"""
    
    def __init__(self, service_name: str, base_url: str):
        self.service_name = service_name
        self.base_url = base_url
        self.client = httpx.AsyncClient(
            timeout=30.0,
            limits=httpx.Limits(max_keepalive_connections=20, max_connections=100)
        )
    
    async def call_service(self, endpoint: str, method: str = "GET", data: Optional[Dict] = None) -> Dict[str, Any]:
        """调用其他微服务"""
        url = f"{self.base_url}{endpoint}"
        
        try:
            if method.upper() == "GET":
                response = await self.client.get(url)
            elif method.upper() == "POST":
                response = await self.client.post(url, json=data)
            elif method.upper() == "PUT":
                response = await self.client.put(url, json=data)
            elif method.upper() == "DELETE":
                response = await self.client.delete(url)
            
            response.raise_for_status()
            return response.json()
            
        except httpx.HTTPStatusError as e:
            raise ServiceCallException(f"服务调用失败: {self.service_name}, 状态码: {e.response.status_code}")
        except httpx.TimeoutException:
            raise ServiceCallException(f"服务调用超时: {self.service_name}")

class ServiceRegistry:
    """服务注册中心"""
    
    def __init__(self):
        self.services = {
            "user-management": ServiceClient("user-management", "http://user-management-service:8002"),
            "content-management": ServiceClient("content-management", "http://content-management-service:8003"),
            "experiment-management": ServiceClient("experiment-management", "http://experiment-management-service:8004"),
            "analytics": ServiceClient("analytics", "http://analytics-service:8005")
        }
    
    def get_service(self, service_name: str) -> ServiceClient:
        """获取服务客户端"""
        if service_name not in self.services:
            raise ValueError(f"未知服务: {service_name}")
        return self.services[service_name]
```

```python
import aio_pika
import json
from typing import Callable, Dict, Any

class MessageQueue:
    """异步消息队列"""
    
    def __init__(self, rabbitmq_url: str = "amqp://rabbitmq-service:5672"):
        self.connection = None
        self.channel = None
        self.rabbitmq_url = rabbitmq_url
        self.message_handlers: Dict[str, Callable] = {}
    
    async def connect(self):
        """连接到RabbitMQ"""
        self.connection = await aio_pika.connect_robust(self.rabbitmq_url)
        self.channel = await self.connection.channel()
        
        # 设置QoS
        await self.channel.set_qos(prefetch_count=10)
    
    async def publish_event(self, event_type: str, data: Dict[str, Any], routing_key: str = ""):
        """发布事件"""
        exchange = await self.channel.declare_exchange("creative_writing_events", aio_pika.ExchangeType.TOPIC)
        
        message_body = json.dumps({
            "event_type": event_type,
            "data": data,
            "timestamp": time.time()
        })
        
        message = aio_pika.Message(
            message_body.encode(),
            headers={"event_type": event_type}
        )
        
        await exchange.publish(message, routing_key=routing_key or event_type)
    
    async def subscribe_to_events(self, event_patterns: List[str], handler: Callable):
        """订阅事件"""
        exchange = await self.channel.declare_exchange("creative_writing_events", aio_pika.ExchangeType.TOPIC)
        queue = await self.channel.declare_queue("", exclusive=True)
        
        for pattern in event_patterns:
            await queue.bind(exchange, routing_key=pattern)
        
        async def on_message(message: aio_pika.IncomingMessage):
            async with message.process():
                try:
                    event_data = json.loads(message.body.decode())
                    await handler(event_data)
                except Exception as e:
                    print(f"事件处理错误: {e}")
        
        await queue.consume(on_message)
```

## 3. 数据流设计

### 3.1 核心数据流路径

#### 3.1.1 用户交互数据流
```python
import asyncio
from dataclasses import dataclass
from typing import List, Dict, Any, Optional
from enum import Enum

class DataFlowStage(Enum):
    INPUT_VALIDATION = "input_validation"
    USER_CONTEXT_LOADING = "user_context_loading"
    AI_INFERENCE = "ai_inference"
    RESULT_PROCESSING = "result_processing"
    RESPONSE_GENERATION = "response_generation"
    DATA_PERSISTENCE = "data_persistence"
    ANALYTICS_UPDATE = "analytics_update"

@dataclass
class UserInteractionFlow:
    """用户交互数据流"""
    session_id: str
    user_id: str
    input_text: str
    timestamp: float
    context_history: List[Dict]
    
    # 流程状态
    current_stage: DataFlowStage = DataFlowStage.INPUT_VALIDATION
    processing_time: Dict[str, float] = None
    errors: List[str] = None
    
    def __post_init__(self):
        if self.processing_time is None:
            self.processing_time = {}
        if self.errors is None:
            self.errors = []

class DataFlowOrchestrator:
    """数据流编排器"""
    
    def __init__(self):
        self.service_registry = ServiceRegistry()
        self.message_queue = MessageQueue()
        self.flow_processors = {
            DataFlowStage.INPUT_VALIDATION: self._validate_input,
            DataFlowStage.USER_CONTEXT_LOADING: self._load_user_context,
            DataFlowStage.AI_INFERENCE: self._execute_ai_inference,
            DataFlowStage.RESULT_PROCESSING: self._process_inference_result,
            DataFlowStage.RESPONSE_GENERATION: self._generate_response,
            DataFlowStage.DATA_PERSISTENCE: self._persist_interaction_data,
            DataFlowStage.ANALYTICS_UPDATE: self._update_analytics
        }
    
    async def process_user_interaction(self, flow: UserInteractionFlow) -> Dict[str, Any]:
        """处理完整的用户交互数据流"""
        start_time = time.time()
        
        try:
            # 按顺序执行各个阶段
            for stage in DataFlowStage:
                stage_start = time.time()
                flow.current_stage = stage
                
                # 执行阶段处理器
                processor = self.flow_processors.get(stage)
                if processor:
                    result = await processor(flow)
                    flow.processing_time[stage.value] = time.time() - stage_start
                    
                    # 检查是否有错误
                    if isinstance(result, dict) and result.get('error'):
                        flow.errors.append(f"{stage.value}: {result['error']}")
                        break
            
            total_time = time.time() - start_time
            
            # 发布完成事件
            await self.message_queue.publish_event(
                EventTypes.USER_INTERACTION,
                {
                    "session_id": flow.session_id,
                    "user_id": flow.user_id,
                    "processing_time": total_time,
                    "stages_completed": len(flow.processing_time),
                    "errors": flow.errors
                }
            )
            
            return {
                "success": len(flow.errors) == 0,
                "processing_time": total_time,
                "stage_times": flow.processing_time,
                "errors": flow.errors
            }
            
        except Exception as e:
            flow.errors.append(f"流程异常: {str(e)}")
            return {"success": False, "error": str(e)}
    
    async def _validate_input(self, flow: UserInteractionFlow) -> Dict[str, Any]:
        """输入验证阶段"""
        try:
            # 基础验证
            if not flow.user_id or not flow.input_text:
                return {"error": "缺少必需的用户ID或输入文本"}
            
            # 输入长度检查
            if len(flow.input_text) > 10000:
                return {"error": "输入文本过长"}
            
            # 内容安全检查
            if self._contains_unsafe_content(flow.input_text):
                return {"error": "输入包含不安全内容"}
            
            return {"success": True}
            
        except Exception as e:
            return {"error": f"输入验证失败: {str(e)}"}
    
    async def _load_user_context(self, flow: UserInteractionFlow) -> Dict[str, Any]:
        """加载用户上下文阶段"""
        try:
            # 调用用户管理服务获取用户信息
            user_service = self.service_registry.get_service("user-management")
            user_info = await user_service.call_service(f"/users/{flow.user_id}")
            
            # 获取用户的学习历史
            learning_history = await user_service.call_service(
                f"/users/{flow.user_id}/learning-history",
                params={"limit": 10}
            )
            
            # 更新流程中的上下文信息
            flow.context_history.extend(learning_history.get("interactions", []))
            
            return {"success": True, "context_loaded": len(flow.context_history)}
            
        except Exception as e:
            return {"error": f"上下文加载失败: {str(e)}"}
    
    async def _execute_ai_inference(self, flow: UserInteractionFlow) -> Dict[str, Any]:
        """AI推理阶段"""
        try:
            # 构建推理请求
            inference_request = {
                "request_id": f"{flow.session_id}_{int(time.time())}",
                "user_id": flow.user_id,
                "input_data": {
                    "text": flow.input_text,
                    "timestamp": flow.timestamp
                },
                "context_history": flow.context_history
            }
            
            # 提交到AI推理服务
            ai_service = self.service_registry.get_service("ai-inference")
            inference_result = await ai_service.call_service(
                "/inference/process",
                method="POST",
                data=inference_request
            )
            
            # 等待推理完成
            request_id = inference_result["request_id"]
            result = await self._wait_for_inference_result(request_id)
            
            # 将结果添加到流程中
            flow.ai_result = result
            
            return {"success": True, "inference_time": result.get("inference_time", 0)}
            
        except Exception as e:
            return {"error": f"AI推理失败: {str(e)}"}
    
    async def _wait_for_inference_result(self, request_id: str, timeout: int = 30) -> Dict[str, Any]:
        """等待推理结果"""
        ai_service = self.service_registry.get_service("ai-inference")
        
        for _ in range(timeout):
            result = await ai_service.call_service(f"/inference/result/{request_id}")
            
            if result["status"] == "completed":
                return result["result"]
            elif result["status"] == "error":
                raise Exception(result.get("error", "推理失败"))
            
            await asyncio.sleep(1)
        
        raise TimeoutError("推理超时")
```

```python
class WebSocketManager:
    """WebSocket连接管理器"""
    
    def __init__(self):
        # 活跃连接: {user_id: {session_id: websocket}}
        self.active_connections: Dict[str, Dict[str, WebSocket]] = {}
        # 房间管理: {room_id: {user_ids}}
        self.rooms: Dict[str, Set[str]] = {}
    
    async def connect(self, websocket: WebSocket, user_id: str, session_id: str):
        """建立WebSocket连接"""
        await websocket.accept()
        
        if user_id not in self.active_connections:
            self.active_connections[user_id] = {}
        
        self.active_connections[user_id][session_id] = websocket
        
        # 发送连接确认
        await self.send_to_user(user_id, {
            "type": "connection_established",
            "session_id": session_id,
            "timestamp": time.time()
        })
    
    def disconnect(self, user_id: str, session_id: str):
        """断开WebSocket连接"""
        if user_id in self.active_connections:
            if session_id in self.active_connections[user_id]:
                del self.active_connections[user_id][session_id]
            
            # 如果用户没有活跃连接，清理用户记录
            if not self.active_connections[user_id]:
                del self.active_connections[user_id]
    
    async def send_to_user(self, user_id: str, message: Dict):
        """向特定用户发送消息"""
        if user_id in self.active_connections:
            disconnected_sessions = []
            
            for session_id, websocket in self.active_connections[user_id].items():
                try:
                    await websocket.send_text(json.dumps(message))
                except:
                    disconnected_sessions.append(session_id)
            
            # 清理断开的连接
            for session_id in disconnected_sessions:
                self.disconnect(user_id, session_id)
    
    async def broadcast_to_room(self, room_id: str, message: Dict, exclude_user: str = None):
        """向房间广播消息"""
        if room_id in self.rooms:
            for user_id in self.rooms[room_id]:
                if exclude_user and user_id == exclude_user:
                    continue
                await self.send_to_user(user_id, message)
    
    async def join_room(self, room_id: str, user_id: str):
        """加入房间"""
        if room_id not in self.rooms:
            self.rooms[room_id] = set()
        
        self.rooms[room_id].add(user_id)
        
        # 通知房间内其他用户
        await self.broadcast_to_room(room_id, {
            "type": "user_joined",
            "user_id": user_id,
            "room_id": room_id
        }, exclude_user=user_id)
```

```python
from enum import Enum
from dataclasses import dataclass
from typing import List, Dict, Any, Optional
import asyncio
import uuid

class TransactionState(Enum):
    PENDING = "pending"
    COMMITTED = "committed"
    ABORTED = "aborted"
    COMPENSATING = "compensating"

@dataclass
class TransactionStep:
    service_name: str
    operation: str
    data: Dict[str, Any]
    compensation_operation: Optional[str] = None
    compensation_data: Optional[Dict[str, Any]] = None

@dataclass
class DistributedTransaction:
    transaction_id: str
    steps: List[TransactionStep]
    state: TransactionState = TransactionState.PENDING
    completed_steps: List[str] = None
    failed_step: Optional[str] = None
    
    def __post_init__(self):
        if self.completed_steps is None:
            self.completed_steps = []

class SagaOrchestrator:
    """Saga模式分布式事务协调器"""
    
    def __init__(self, service_registry: ServiceRegistry):
        self.service_registry = service_registry
        self.active_transactions: Dict[str, DistributedTransaction] = {}
    
    async def execute_transaction(self, steps: List[TransactionStep]) -> bool:
        """执行分布式事务"""
        transaction_id = str(uuid.uuid4())
        transaction = DistributedTransaction(
            transaction_id=transaction_id,
            steps=steps
        )
        
        self.active_transactions[transaction_id] = transaction
        
        try:
            # 执行前向流程
            success = await self._execute_forward_flow(transaction)
            
            if success:
                transaction.state = TransactionState.COMMITTED
                return True
            else:
                # 执行补偿流程
                await self._execute_compensation_flow(transaction)
                transaction.state = TransactionState.ABORTED
                return False
                
        except Exception as e:
            print(f"事务执行异常: {e}")
            await self._execute_compensation_flow(transaction)
            transaction.state = TransactionState.ABORTED
            return False
        
        finally:
            # 清理事务记录
            if transaction_id in self.active_transactions:
                del self.active_transactions[transaction_id]
    
    async def _execute_forward_flow(self, transaction: DistributedTransaction) -> bool:
        """执行前向流程"""
        for i, step in enumerate(transaction.steps):
            try:
                service = self.service_registry.get_service(step.service_name)
                result = await service.call_service(
                    step.operation,
                    method="POST",
                    data=step.data
                )
                
                if result.get("success", False):
                    transaction.completed_steps.append(f"step_{i}")
                else:
                    transaction.failed_step = f"step_{i}"
                    return False
                    
            except Exception as e:
                transaction.failed_step = f"step_{i}"
                print(f"事务步骤 {i} 执行失败: {e}")
                return False
        
        return True
    
    async def _execute_compensation_flow(self, transaction: DistributedTransaction):
        """执行补偿流程"""
        transaction.state = TransactionState.COMPENSATING
        
        # 逆序执行补偿操作
        for i in reversed(range(len(transaction.completed_steps))):
            step = transaction.steps[i]
            
            if step.compensation_operation:
                try:
                    service = self.service_registry.get_service(step.service_name)
                    await service.call_service(
                        step.compensation_operation,
                        method="POST",
                        data=step.compensation_data or {}
                    )
                except Exception as e:
                    print(f"补偿操作 {i} 执行失败: {e}")

# 使用示例：用户注册分布式事务
async def create_user_with_transaction(user_data: Dict[str, Any]) -> bool:
    """创建用户的分布式事务"""
    orchestrator = SagaOrchestrator(service_registry)
    
    steps = [
        # 步骤1：在用户管理服务创建用户

            service_name="user-management",
            operation="/users",
            data=user_data,

            data={"user_id": user_data["user_id"], "experiment_id": user_data["experiment_id"]},
            compensation_operation="/experiments/participants/{user_id}/remove",
            compensation_data={"user_id": user_data["user_id"]}
        ),
        
        # 步骤3：初始化用户分析数据
        TransactionStep(

            data={"user_id": user_data["user_id"]},
            compensation_operation="/analytics/users/{user_id}/cleanup",
```

#### 5.2.1 多级缓存设计
```python
import asyncio
import json
from typing import Optional, Any, Dict
from dataclasses import dataclass
import time

@dataclass
class CacheItem:
    key: str
    value: Any
    ttl: int
    created_at: float
    access_count: int = 0
    last_accessed: float = 0

class MultiLevelCache:
    """多级缓存系统"""
    
    def __init__(self, redis_client, local_cache_size: int = 1000):
        self.redis_client = redis_client
        self.local_cache: Dict[str, CacheItem] = {}
        self.local_cache_size = local_cache_size
    
    async def get(self, key: str) -> Optional[Any]:
        """获取缓存值"""
        # L1: 本地内存缓存
        if key in self.local_cache:
            item = self.local_cache[key]
            
            # 检查是否过期
            if time.time() - item.created_at < item.ttl:
                item.access_count += 1
                item.last_accessed = time.time()
                return item.value
            else:
                del self.local_cache[key]
        
        # L2: Redis缓存
        try:
            redis_value = await self.redis_client.get(key)
            if redis_value:
                value = json.loads(redis_value)
                
                # 回写到本地缓存
                await self._set_local_cache(key, value, ttl=300)
                
                return value
        except Exception as e:
            print(f"Redis缓存获取失败: {e}")
        
        return None
    
    async def set(self, key: str, value: Any, ttl: int = 3600):
        """设置缓存值"""
        # 同时设置两级缓存
        await self._set_local_cache(key, value, ttl)
        await self._set_redis_cache(key, value, ttl)
    
    async def _set_local_cache(self, key: str, value: Any, ttl: int):
        """设置本地缓存"""
        # LRU淘汰策略
        if len(self.local_cache) >= self.local_cache_size:
            # 找到最少使用的项
            lru_key = min(
                self.local_cache.keys(),
                key=lambda k: self.local_cache[k].last_accessed
            )
            del self.local_cache[lru_key]
        
        self.local_cache[key] = CacheItem(
            key=key,
            value=value,
            ttl=ttl,
            created_at=time.time(),
            last_accessed=time.time()
        )
    
    async def _set_redis_cache(self, key: str, value: Any, ttl: int):
        """设置Redis缓存"""
        try:
            await self.redis_client.setex(
                key, 
                ttl, 
                json.dumps(value, ensure_ascii=False)
            )
        except Exception as e:
            print(f"Redis缓存设置失败: {e}")

            result = await func(*args, **kwargs)
            
            # 缓存结果
            await cache.set(cache_key, result, ttl)
            
            return result
        
        return wrapper
    return decorator
```

```python
        self.secret_key = secret_key
        self.algorithm = algorithm
        self.access_token_expire = timedelta(hours=2)
        self.refresh_token_expire = timedelta(days=7)
    
    def create_access_token(self, user_id: str, permissions: list = None) -> str:
        """创建访问令牌"""
        payload = {
            "user_id": user_id,
            "permissions": permissions or [],
            "type": "access",
            "exp": datetime.utcnow() + self.access_token_expire,
            "iat": datetime.utcnow()
        }
        
        return jwt.encode(payload, self.secret_key, algorithm=self.algorithm)
    
    def create_refresh_token(self, user_id: str) -> str:
        """创建刷新令牌"""
        payload = {
            "user_id": user_id,
            "type": "refresh",
            "exp": datetime.utcnow() + self.refresh_token_expire,
            "iat": datetime.utcnow()
        }
        
        return jwt.encode(payload, self.secret_key, algorithm=self.algorithm)
    
    def verify_token(self, token: str) -> Dict[str, Any]:
        """验证令牌"""
        try:
            payload = jwt.decode(token, self.secret_key, algorithms=[self.algorithm])
            return payload
        except jwt.ExpiredSignatureError:
            raise HTTPException(status_code=401, detail="令牌已过期")
        except jwt.JWTError:
            raise HTTPException(status_code=401, detail="无效令牌")

security = HTTPBearer()
jwt_manager = JWTManager(secret_key="your-secret-key")

async def get_current_user(credentials: HTTPAuthorizationCredentials = Depends(security)):
    """获取当前用户"""
    token = credentials.credentials
    payload = jwt_manager.verify_token(token)
    
    if payload["type"] != "access":
        raise HTTPException(status_code=401, detail="无效的访问令牌")
    
    return {
        "user_id": payload["user_id"],
        "permissions": payload["permissions"]
    }

def require_permission(permission: str):
    """权限检查装饰器"""
    def decorator(func):
        async def wrapper(*args, **kwargs):
            # 从依赖注入中获取当前用户
            current_user = kwargs.get("current_user")
            if not current_user:
                raise HTTPException(status_code=401, detail="未认证")
            
            if permission not in current_user["permissions"]:
                raise HTTPException(status_code=403, detail="权限不足")
            
            return await func(*args, **kwargs)
        return wrapper
    return decorator
```

```python
class DataEncryption:
    """数据加密服务"""
    
    def __init__(self, master_key: str):
        self.master_key = master_key.encode()
        self.fernet = self._create_fernet()
    
    def _create_fernet(self) -> Fernet:
        """创建Fernet加密器"""
        salt = b'creative_writing_salt'  # 在生产环境中应该是随机的
        kdf = PBKDF2HMAC(
            algorithm=hashes.SHA256(),
            length=32,
            salt=salt,
            iterations=100000,
        )
        key = base64.urlsafe_b64encode(kdf.derive(self.master_key))
        return Fernet(key)
    
    def encrypt_text(self, plaintext: str) -> str:
        """加密文本"""
        encrypted_data = self.fernet.encrypt(plaintext.encode())
        return base64.urlsafe_b64encode(encrypted_data).decode()
    
    def decrypt_text(self, encrypted_text: str) -> str:
        """解密文本"""
        encrypted_data = base64.urlsafe_b64decode(encrypted_text.encode())
        decrypted_data = self.fernet.decrypt(encrypted_data)
        return decrypted_data.decode()
    
    def encrypt_user_data(self, user_data: Dict[str, Any]) -> Dict[str, Any]:
        """加密用户敏感数据"""

        encrypted_data = user_data.copy()
        
        for field in sensitive_fields:
            if field in encrypted_data:
                encrypted_data[field] = self.encrypt_text(str(encrypted_data[field]))
        
        return encrypted_data

                except Exception as e:
                    print(f"解密字段 {field} 失败: {e}")
```
