## 1. 知识图谱总体设计

### 1.1 图谱概述

创意写作知识图谱是AI系统的核心知识基础，旨在构建一个覆盖创意写作全流程的专业知识网络，支撑GraphSAGE算法的深度推理。

### 1.2 设计原则

1. **领域专业性**: 专注创意写作教学领域，确保知识的专业性和准确性
2. **层次结构化**: 建立清晰的概念层次，从抽象理论到具体技法
3. **动态可扩展**: 支持在线学习和知识更新，适应用户个性化需求
4. **推理友好**: 优化图结构，支持高效的图神经网络推理

### 1.3 图谱规模目标

```python
    def __init__(self):
        self.graph = nx.MultiDiGraph()

    def add_concept_node(self, name, description, attributes=None):
        """添加概念节点"""
        node_id = f"concept_{self.node_counter}"
        self.node_counter += 1
        
        node_attrs = {
            'id': node_id,
            'name': name,
            'description': description,
            'type': NodeType.CONCEPT,
            'create_time': datetime.now()
        }
        
        if attributes:
            node_attrs.update(attributes)
            
        # 添加到NetworkX图
        self.graph.add_node(node_id, **node_attrs)
        
        # 添加到RDF图
        concept_uri = URIRef(self.namespace + node_id)
        self.rdf_graph.add((concept_uri, RDF.type, OWL.Class))
        self.rdf_graph.add((concept_uri, RDFS.label, Literal(name)))
        self.rdf_graph.add((concept_uri, RDFS.comment, Literal(description)))
        
        return node_id
    
    def add_technique_node(self, name, steps, applicable_genres, attributes=None):
        """添加技法节点"""

        node_attrs = {
            'id': node_id,
            'name': name,
            'type': NodeType.TECHNIQUE,
            'implementation_steps': steps,
            'applicable_genres': applicable_genres,
            'create_time': datetime.now()
        }
        
        if attributes:
            node_attrs.update(attributes)
            
        self.graph.add_node(node_id, **node_attrs)
        
        # RDF表示
        technique_uri = URIRef(self.namespace + node_id)
        self.rdf_graph.add((technique_uri, RDF.type, self.namespace.Technique))
        self.rdf_graph.add((technique_uri, RDFS.label, Literal(name)))
        
        return node_id
    
    def add_relation(self, source_id, target_id, relation_type, attributes=None):
        """添加关系"""
        relation_attrs = {
            'type': relation_type,
            'weight': 1.0,
            'confidence': 0.8,
            'create_time': datetime.now()
        }
        
        if attributes:
            relation_attrs.update(attributes)
            
        # 添加到NetworkX图
        self.graph.add_edge(source_id, target_id, **relation_attrs)
        
        # 添加到RDF图
        source_uri = URIRef(self.namespace + source_id)
        target_uri = URIRef(self.namespace + target_id)
        relation_uri = URIRef(self.namespace + relation_type.value)
        
        self.rdf_graph.add((source_uri, relation_uri, target_uri))
        
        return f"{source_id}_{relation_type.value}_{target_id}"

        # 3. 添加具体技法
        flashback = self.add_technique_node(

        self.add_relation(

        # 统计关系类型

            rel_type = attrs.get('type', 'unknown')
            stats['relation_types'][rel_type] = stats['relation_types'].get(rel_type, 0) + 1
```

```python
import spacy
from transformers import AutoTokenizer, AutoModelForTokenClassification
from transformers import pipeline

class CreativeWritingNER:
    def __init__(self):
        # 加载预训练模型
        self.nlp = spacy.load("zh_core_web_sm")
        
        # 初始化BERT-based NER模型
        self.tokenizer = AutoTokenizer.from_pretrained("hfl/chinese-bert-wwm-ext")
        self.model = AutoModelForTokenClassification.from_pretrained("hfl/chinese-bert-wwm-ext")
        self.ner_pipeline = pipeline("ner", 
                                   model=self.model, 
                                   tokenizer=self.tokenizer,
                                   aggregation_strategy="simple")
        
        # 自定义实体类型
        self.entity_types = {
            'WRITING_TECHNIQUE': '写作技法',
            'LITERARY_CONCEPT': '文学概念',
            'TEXT_ELEMENT': '文本元素',
            'AUTHOR': '作者',
            'WORK': '作品',
            'GENRE': '文学体裁',
            'RHETORICAL_DEVICE': '修辞手法'
        }
        
    def extract_entities(self, text):
        """抽取命名实体"""
        # 1. 使用spaCy进行基础NER
        doc = self.nlp(text)
        spacy_entities = [(ent.text, ent.label_) for ent in doc.ents]
        
        # 2. 使用BERT模型进行深度NER
        bert_entities = self.ner_pipeline(text)
        
        # 3. 自定义规则匹配
        custom_entities = self._extract_custom_entities(text)
        
        # 4. 实体融合和去重
        merged_entities = self._merge_entities(spacy_entities, bert_entities, custom_entities)
        
        return merged_entities
    
    def _extract_custom_entities(self, text):
        """自定义实体抽取规则"""
        custom_entities = []
        
        # 写作技法模式
        technique_patterns = [
            r"(倒叙|插叙|顺叙)法",
            r"(对比|排比|比喻|拟人)手法",
            r"(悬念|伏笔|照应)技巧",
            r"(第一人称|第三人称)叙述"
        ]
        
        for pattern in technique_patterns:
            matches = re.finditer(pattern, text)
            for match in matches:
                custom_entities.append({
                    'text': match.group(),
                    'type': 'WRITING_TECHNIQUE',
                    'start': match.start(),
                    'end': match.end(),
                    'confidence': 0.9
                })
        
        return custom_entities
    
    def _merge_entities(self, spacy_ents, bert_ents, custom_ents):
        """融合多种来源的实体"""
        all_entities = []
        
        # 处理spaCy实体
        for text, label in spacy_ents:
            all_entities.append({
                'text': text,
                'type': label,
                'source': 'spacy',
                'confidence': 0.8
            })
        
        # 处理BERT实体
        for ent in bert_ents:
            all_entities.append({
                'text': ent['word'],
                'type': ent['entity_group'],
                'source': 'bert',
                'confidence': ent['score']
            })
        
        # 添加自定义实体
        all_entities.extend(custom_ents)
        
        # 去重和融合
        merged = self._deduplicate_entities(all_entities)
        
        return merged
    
    def _deduplicate_entities(self, entities):
        """实体去重"""
        # 按文本内容分组
        entity_groups = {}
        for ent in entities:
            text = ent['text'].strip()
            if text not in entity_groups:
                entity_groups[text] = []
            entity_groups[text].append(ent)
        
        # 选择最佳实体
        deduplicated = []
        for text, group in entity_groups.items():
            if len(group) == 1:
                deduplicated.append(group[0])
            else:
                # 选择置信度最高的
                best_entity = max(group, key=lambda x: x['confidence'])
                deduplicated.append(best_entity)
        
        return deduplicated
```

```python
class RelationExtractor:
    def __init__(self):
        self.relation_patterns = self._load_relation_patterns()
        self.relation_classifier = self._load_relation_classifier()
        
    def extract_relations(self, text, entities):
        """抽取实体间关系"""
        relations = []
        
        # 1. 基于模式的关系抽取
        pattern_relations = self._pattern_based_extraction(text, entities)
        relations.extend(pattern_relations)
        
        # 2. 基于机器学习的关系分类
        ml_relations = self._ml_based_extraction(text, entities)
        relations.extend(ml_relations)
        
        # 3. 基于句法分析的关系抽取
        syntax_relations = self._syntax_based_extraction(text, entities)
        relations.extend(syntax_relations)
        
        # 4. 关系过滤和验证
        validated_relations = self._validate_relations(relations)
        
        return validated_relations
    
    def _pattern_based_extraction(self, text, entities):
        """基于模式的关系抽取"""
        relations = []
        
        # 定义关系模式
        relation_patterns = {
            RelationType.IS_A: [
                r"(.+)是(.+)的一种",
                r"(.+)属于(.+)",
                r"(.+)是(.+)"
            ],
            RelationType.PART_OF: [
                r"(.+)包含(.+)",
                r"(.+)由(.+)组成",
                r"(.+)的(.+)"
            ],
            RelationType.APPLIES_TO: [
                r"(.+)用于(.+)",
                r"(.+)适用于(.+)",
                r"在(.+)中使用(.+)"
            ]
        }
        
        for relation_type, patterns in relation_patterns.items():
            for pattern in patterns:
                matches = re.finditer(pattern, text)
                for match in matches:
                    entity1 = match.group(1).strip()
                    entity2 = match.group(2).strip()
                    
                    # 验证实体是否在识别列表中
                    if self._entity_exists(entity1, entities) and self._entity_exists(entity2, entities):
                        relations.append({
                            'source': entity1,
                            'target': entity2,
                            'type': relation_type,
                            'confidence': 0.8,
                            'evidence': match.group(),
                            'method': 'pattern'
                        })
        
        return relations
    
    def _ml_based_extraction(self, text, entities):
        """基于机器学习的关系抽取"""
        relations = []
        
        # 生成实体对
        entity_pairs = []
        for i, ent1 in enumerate(entities):
            for j, ent2 in enumerate(entities[i+1:], i+1):
                if ent1['text'] != ent2['text']:
                    entity_pairs.append((ent1, ent2))
        
        # 对每个实体对进行关系分类
        for ent1, ent2 in entity_pairs:
            # 提取特征
            features = self._extract_relation_features(text, ent1, ent2)
            
            # 关系分类
            relation_type, confidence = self.relation_classifier.predict(features)
            
            if confidence > 0.6:  # 置信度阈值
                relations.append({
                    'source': ent1['text'],
                    'target': ent2['text'],
                    'type': relation_type,
                    'confidence': confidence,
                    'method': 'ml'
                })
        
        return relations
    
    def _extract_relation_features(self, text, ent1, ent2):
        """提取关系特征"""
        features = {}
        
        # 1. 词汇特征
        words_between = self._get_words_between(text, ent1, ent2)
        features['words_between'] = words_between
        features['distance'] = len(words_between)
        
        # 2. 句法特征
        features['pos_tags'] = self._get_pos_tags(words_between)
        features['dependency_path'] = self._get_dependency_path(text, ent1, ent2)
        
        # 3. 语义特征
        features['semantic_similarity'] = self._compute_semantic_similarity(ent1['text'], ent2['text'])
        
        # 4. 实体类型特征
        features['ent1_type'] = ent1.get('type', 'UNK')
        features['ent2_type'] = ent2.get('type', 'UNK')
        
        return features
    
    def _load_relation_classifier(self):
        """加载关系分类器"""
        # 这里可以使用预训练的BERT模型或传统ML模型
        # 为简化，使用基于规则的分类器
        return RuleBasedRelationClassifier()

class RuleBasedRelationClassifier:
    """基于规则的关系分类器"""
    def __init__(self):
        self.rules = {
            ('WRITING_TECHNIQUE', 'TEXT_ELEMENT'): RelationType.APPLIES_TO,
            ('LITERARY_CONCEPT', 'WRITING_TECHNIQUE'): RelationType.IS_A,
            ('AUTHOR', 'WORK'): RelationType.CREATED,
            ('WORK', 'GENRE'): RelationType.BELONGS_TO
        }
    
    def predict(self, features):
        """预测关系类型"""
        ent1_type = features.get('ent1_type')
        ent2_type = features.get('ent2_type')
        
        # 查找匹配规则
        rule_key = (ent1_type, ent2_type)
        if rule_key in self.rules:
            return self.rules[rule_key], 0.8
        
        # 默认关系
        return RelationType.ASSOCIATED_WITH, 0.5
```

```python
class CompletenessEvaluator:

    def evaluate(self, kg):

class ConsistencyEvaluator:
    """一致性评估器"""
    def evaluate(self, kg):
        """评估知识图谱的一致性"""
        inconsistencies = []
        
        # 1. 检查循环依赖
        cycles = self._detect_cycles(kg)
        inconsistencies.extend(cycles)
        
        # 2. 检查关系冲突
        conflicts = self._detect_relation_conflicts(kg)
        inconsistencies.extend(conflicts)
        
        # 3. 检查属性一致性
        attr_inconsistencies = self._check_attribute_consistency(kg)
        inconsistencies.extend(attr_inconsistencies)
        
        # 4. 检查类型一致性
        type_inconsistencies = self._check_type_consistency(kg)
        inconsistencies.extend(type_inconsistencies)
        
        # 计算一致性分数
        total_nodes = kg.number_of_nodes()
        inconsistency_rate = len(inconsistencies) / max(total_nodes, 1)
        consistency_score = max(0, 1 - inconsistency_rate)
        
        return consistency_score, {
            'inconsistencies': inconsistencies,
            'inconsistency_count': len(inconsistencies),
            'inconsistency_rate': inconsistency_rate
        }
    
    def _detect_cycles(self, kg):
        """检测循环依赖"""
        cycles = []
        
        # 构建层次关系子图
        hierarchy_graph = nx.DiGraph()
        for source, target, attrs in kg.edges(data=True):
            rel_type = attrs.get('type')
            if rel_type in [RelationType.IS_A, RelationType.PART_OF]:
                hierarchy_graph.add_edge(source, target)
        
        # 检测强连通组件
        try:
            cycle_list = list(nx.simple_cycles(hierarchy_graph))
            for cycle in cycle_list:
                cycles.append({
                    'type': 'hierarchy_cycle',
                    'nodes': cycle,
                    'severity': 'high'
                })
        except:
            pass
        
        return cycles
    
    def _detect_relation_conflicts(self, kg):
        """检测关系冲突"""
        conflicts = []
        
        # 检查互斥关系
        conflict_pairs = [
            (RelationType.IS_A, RelationType.PART_OF),
            (RelationType.APPLIES_TO, RelationType.CONFLICTS_WITH)
        ]
        
        for node_pair in kg.edges():
            source, target = node_pair
            edge_data = kg.get_edge_data(source, target)
            
            if edge_data and len(edge_data) > 1:  # 多重边
                relations = [attrs.get('type') for attrs in edge_data.values()]
                
                for rel1, rel2 in conflict_pairs:
                    if rel1 in relations and rel2 in relations:
                        conflicts.append({
                            'type': 'relation_conflict',
                            'source': source,
                            'target': target,
                            'conflicting_relations': [rel1, rel2],
                            'severity': 'medium'
                        })
        
        return conflicts

        invalid_chars = ['/', '\\', '?', '*', '|', '<', '>']
        if any(char in name for char in invalid_chars):
            return False
        
        # 检查长度合理性
        if len(name) > 100:
            return False
        
        return True

        matched_keywords = sum(1 for keyword in core_keywords
```

```python
    """为GraphSAGE准备训练数据"""
    def __init__(self, neo4j_kg):

    def extract_subgraphs_for_training(self, num_subgraphs=1000, max_nodes_per_subgraph=500):
        """提取训练用子图"""
        subgraphs = []
        
        with self.kg.driver.session() as session:
            for i in range(num_subgraphs):
                # 随机选择种子节点
                seed_node = session.run("""
                    MATCH (n)

                    ORDER BY rand()
                    LIMIT 1
                """).single()['id']
                
                # 提取以种子节点为中心的子图

                subgraphs.append(subgraph)
        
        return subgraphs
    
    def _extract_k_hop_subgraph(self, session, seed_node, k=3, max_nodes=500):
        """提取k-hop子图"""
        result = session.run("""
            MATCH path = (seed {id: $seed_id})-[*1..$k]-(neighbor)
            WITH collect(DISTINCT seed) + collect(DISTINCT neighbor) as nodes,
                 collect(DISTINCT relationships(path)) as all_rels
            UNWIND all_rels as rel_list
            UNWIND rel_list as rel
            WITH nodes, collect(DISTINCT rel) as relationships
            RETURN nodes[0..$max_nodes] as limited_nodes, relationships
        """, seed_id=seed_node, k=k, max_nodes=max_nodes)
        
        subgraph_data = result.single()
        return {
            'nodes': subgraph_data['limited_nodes'],
            'relationships': subgraph_data['relationships']
        }

                RETURN n.id as id, n.name as name, labels(n)[0] as type,
                       n.difficulty as difficulty, n.cognitive_load as cognitive_load,
                       n.innovation as innovation
                ORDER BY n.id
            """)
            
            nodes = list(nodes_result)
            node_mapping = {node['id']: i for i, node in enumerate(nodes)}
            
            # 构建节点特征矩阵
            node_features = []
            for node in nodes:
                features = [
                    node['difficulty'] or 0,
                    node['cognitive_load'] or 0,
                    node['innovation'] or 0
                ]
                node_features.append(features)
            
            x = torch.tensor(node_features, dtype=torch.float)
            
            # 获取边信息
            edges_result = session.run("""
                MATCH (source)-[r]->(target)
                RETURN source.id as source_id, target.id as target_id,
                       type(r) as rel_type, r.weight as weight
            """)
            
            edge_index = []
            edge_attr = []
            
            for edge in edges_result:
                source_idx = node_mapping[edge['source_id']]
                target_idx = node_mapping[edge['target_id']]
                
                edge_index.append([source_idx, target_idx])
                edge_attr.append([edge['weight'] or 1.0])
            
            edge_index = torch.tensor(edge_index, dtype=torch.long).t().contiguous()
            edge_attr = torch.tensor(edge_attr, dtype=torch.float)
            
            # 创建PyTorch Geometric数据对象
            data = Data(x=x, edge_index=edge_index, edge_attr=edge_attr)
            
            # 保存数据
            torch.save(data, output_path)
            print(f"PyTorch Geometric数据已保存到 {output_path}")
            
            return data
```

```python
        schedule.every().week.do(self._rebuild_indexes)
        
        # 每月任务

        with self.kg.driver.session() as session:

                MATCH (n)
                SET n.last_updated = datetime(),
                    n.degree = size((n)--()),
                    n.in_degree = size((n)<--()),
                    n.out_degree = size((n)-->())
            """)
```
